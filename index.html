펌웨어 올리는 방법

  1. VS Code와 ESP-IDF 설치VS Code에서 “Espressif IDF” 확장을 설치하고, 설정 마법사에서 ESP-IDF v5.4 이상을 설치합니다.
  2. 프로젝트 열기내려받은 펌웨어 폴더를 VS Code로 열고, 아래쪽 상태 표시줄에서 대상 칩을 esp32c3로 선택합니다.
  3. USB-C로 연결상태 표시줄에서 포트를 고릅니다. 포트가 보이지 않으면 보드의 B(BOOT) 버튼을 누른 채 USB를 꽂고, 연결 후 버튼을 뗍니다.
  4. 빌드 · 업로드상태 표시줄의 Build, Flash를 차례로 누릅니다. 터미널에서는 idf.py set-target esp32c3 다음에 idf.py build flash monitor를 실행해도 됩니다.

예제 코드

XIAO ESP32C3, SSD1306 호환 OLED(128×64, I2C), ESP-IDF v5.4 이상 기준입니다. SDA는 D4, SCL은 D5에 연결합니다.

LESSON 1

화면에 눈 두 개 띄우기

1비트 프레임버퍼에 둥근 사각형 두 개를 그려 OLED로 보냅니다. 크기와 위치를 바꿔 보며 좌표 감각을 익힙니다. 코드는 main/main.c에 넣습니다.

#include <string.h>
#include "driver/i2c_master.h"
#include "esp_lcd_panel_io.h"
#include "esp_lcd_panel_ops.h"
#include "esp_lcd_panel_vendor.h"
#include "esp_random.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define SDA_GPIO 6   // XIAO ESP32C3 D4
#define SCL_GPIO 7   // XIAO ESP32C3 D5
#define W 128
#define H 64

static uint8_t fb[W * H / 8];          // 1비트 프레임버퍼
static esp_lcd_panel_handle_t panel;

static void px(int x, int y) {
    if (x < 0 || x >= W || y < 0 || y >= H) return;
    fb[(y >> 3) * W + x] |= 1 << (y & 7);
}

static void rbox(int x, int y, int w, int h, int r) {  // 둥근 사각형
    if (r > h / 2) r = h / 2;
    for (int j = 0; j < h; j++) for (int i = 0; i < w; i++) {
        int dx = i < r ? r - 1 - i : (i >= w - r ? i - (w - r) : 0);
        int dy = j < r ? r - 1 - j : (j >= h - r ? j - (h - r) : 0);
        if (dx * dx + dy * dy <= r * r) px(x + i, y + j);
    }
}

static void draw_eyes(int h) {
    memset(fb, 0, sizeof(fb));
    int y = 32 - h / 2;
    rbox(29, y, 22, h, 8);             // 왼쪽 눈
    rbox(77, y, 22, h, 8);             // 오른쪽 눈
    esp_lcd_panel_draw_bitmap(panel, 0, 0, W, H, fb);
}

static void oled_init(void) {
    i2c_master_bus_config_t bus_cfg = {
        .clk_source = I2C_CLK_SRC_DEFAULT, .i2c_port = I2C_NUM_0,
        .sda_io_num = SDA_GPIO, .scl_io_num = SCL_GPIO,
        .glitch_ignore_cnt = 7, .flags.enable_internal_pullup = true,
    };
    i2c_master_bus_handle_t bus;
    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_cfg, &bus));

    esp_lcd_panel_io_i2c_config_t io_cfg = {
        .dev_addr = 0x3C, .scl_speed_hz = 400000,
        .control_phase_bytes = 1, .dc_bit_offset = 6,
        .lcd_cmd_bits = 8, .lcd_param_bits = 8,
    };
    esp_lcd_panel_io_handle_t io;
    ESP_ERROR_CHECK(esp_lcd_new_panel_io_i2c(bus, &io_cfg, &io));

    esp_lcd_panel_ssd1306_config_t ssd = { .height = H };
    esp_lcd_panel_dev_config_t cfg = {
        .bits_per_pixel = 1, .reset_gpio_num = -1, .vendor_config = &ssd,
    };
    ESP_ERROR_CHECK(esp_lcd_new_panel_ssd1306(io, &cfg, &panel));
    esp_lcd_panel_reset(panel);
    esp_lcd_panel_init(panel);
    esp_lcd_panel_disp_on_off(panel, true);
}

void app_main(void) {
    oled_init();
    draw_eyes(28);
}
LESSON 2

눈 깜빡이기

눈의 높이를 줄였다 늘리면 깜빡임이 됩니다. 간격을 무작위로 주면 살아 있는 느낌이 납니다. 레슨 1의 app_main을 아래로 바꿉니다.

void app_main(void) {
    oled_init();
    while (1) {
        for (int h = 28; h >= 2; h -= 6) { draw_eyes(h); vTaskDelay(pdMS_TO_TICKS(15)); }
        for (int h = 2; h <= 28; h += 6)  { draw_eyes(h); vTaskDelay(pdMS_TO_TICKS(15)); }
        draw_eyes(28);
        vTaskDelay(pdMS_TO_TICKS(1500 + esp_random() % 2500));
    }
}
